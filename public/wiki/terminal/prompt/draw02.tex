   \ifnum #1<10 00#1 \else \ifnum #1<100 0#1 \else #1 \fi \fi
}
\newcommand{\colorcell}[3]{
  \def\x{#1}
  \def\y{#2}
  \def\c{#3}

  \fill[fill=xterm\c] (\x,\y + 0.1) rectangle (\x + 0.5,\y + 0.7);
  \node[text=xterm\c, font=\fontsize{7pt}{\baselineskip}\selectfont, align=right] at (\x + 1.2,\y + 0.4) {\ttfamily\bfseries\twodigits{\c}};
}
