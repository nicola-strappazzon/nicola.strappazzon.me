```C
/*
 * LCD 16x2 (HD44780) via PCF8574 I2C backpack — AVR128DA28.
 *
 * Pinout:
 *   PC2 TWI0 SDA (ALT2)  → PCF8574 SDA / LCD SDA
 *   PC3 TWI0 SCL (ALT2)  → PCF8574 SCL / LCD SCL
 *
 * I2C address: 0x27 by default. Try 0x3F if your module does not answer.
 */

#include <avr/io.h>
#include <util/delay.h>
#include <stdbool.h>
#include <stdint.h>
#include <stdio.h>

/* ============================================================================
 * TWI0 master @ 100 kHz, ALT2 (PC2/PC3) at F_CPU = 24 MHz
 *
 * MBAUD = F_CPU/(2*F_SCL) - 5 = 24M/(2*100k) - 5 = 115
 * ========================================================================== */

static void TWI0_init(void) {
    PORTMUX.TWIROUTEA = 0x02;
    TWI0.CTRLA    = TWI_SDAHOLD_50NS_gc;
    TWI0.DUALCTRL = 0x00;
    TWI0.DBGCTRL  = TWI_DBGRUN_bm;
    TWI0.MBAUD    = 115;
    TWI0.MSTATUS  = TWI_RIF_bm | TWI_WIF_bm | TWI_CLKHOLD_bm |
                    TWI_RXACK_bm | TWI_ARBLOST_bm | TWI_BUSERR_bm |
                    TWI_BUSSTATE_IDLE_gc;
    TWI0.MCTRLA   = TWI_ENABLE_bm;
}

static bool TWI_write(uint8_t addr, const uint8_t *data, uint16_t len) {
    TWI0.MADDR = (addr << 1) | 0;
    for (uint16_t i = 0; i < len; i++) {
        while (!(TWI0.MSTATUS & TWI_WIF_bm));
        if (TWI0.MSTATUS & (TWI_ARBLOST_bm | TWI_BUSERR_bm)) return false;
        if (TWI0.MSTATUS & TWI_RXACK_bm)                     return false;
        TWI0.MDATA = data[i];
    }
    while (!(TWI0.MSTATUS & TWI_WIF_bm));
    TWI0.MCTRLB = TWI_MCMD_STOP_gc;
    return true;
}

/* ============================================================================
 * HD44780 LCD via PCF8574 — 4-bit mode
 *
 * PCF8574 byte layout:
 *   bit 7..4 = D7..D4 (data nibble)
 *   bit 3    = backlight (1 = on)
 *   bit 2    = EN
 *   bit 1    = RW (always 0)
 *   bit 0    = RS (0 = command, 1 = data)
 *
 * Each LCD byte is sent as two PCF8574 nibbles (upper first), and each
 * nibble is latched on the falling edge of EN.
 * ========================================================================== */

#define LCD_ADDR        0x27        /* try 0x3F if your module is silent */
#define LCD_RS          0x01
#define LCD_EN          0x04
#define LCD_BL          0x08
#define LCD_CLEAR       0x01
#define LCD_ENTRY_MODE  0x04        /* | 0x02 = increment | 0x01 = shift */
#define LCD_DISPLAY     0x08        /* | 0x04 = on | 0x02 = cursor | 0x01 = blink */
#define LCD_FUNCTION    0x20        /* | 0x10 = 8-bit | 0x08 = 2 lines | 0x04 = 5x10 */
#define LCD_SET_CGRAM   0x40
#define LCD_SET_DDRAM   0x80

static void lcd_write_pcf(uint8_t value) {
    TWI_write(LCD_ADDR, &value, 1);
}

static void lcd_pulse_en(uint8_t value) {
    lcd_write_pcf(value | LCD_EN);
    _delay_us(1);
    lcd_write_pcf(value & ~LCD_EN);
    _delay_us(50);
}

static void lcd_write_nibble(uint8_t nibble, uint8_t rs) {
    uint8_t v = (uint8_t)((nibble & 0xF0) | rs | LCD_BL);
    lcd_write_pcf(v);
    lcd_pulse_en(v);
}

static void lcd_write_byte(uint8_t value, uint8_t rs) {
    lcd_write_nibble((uint8_t)(value & 0xF0), rs);
    lcd_write_nibble((uint8_t)(value << 4),   rs);
}

static void lcd_command(uint8_t cmd) {
    lcd_write_byte(cmd, 0);
    if (cmd == LCD_CLEAR) _delay_ms(2);
}

static void lcd_data(uint8_t data) {
    lcd_write_byte(data, LCD_RS);
}

/* HD44780 datasheet §24 — 4-bit init sequence. The chip wakes up in 8-bit
 * mode, so we send 0x3 three times with the mandatory delays, then switch
 * to 4-bit (0x2) before issuing real commands. */
static void LCD_init(void) {
    _delay_ms(50);                       /* > 40 ms after Vcc ≥ 2.7 V */
    lcd_write_pcf(LCD_BL);
    _delay_ms(1);

    lcd_write_nibble(0x30, 0); _delay_ms(5);
    lcd_write_nibble(0x30, 0); _delay_us(150);
    lcd_write_nibble(0x30, 0); _delay_us(150);
    lcd_write_nibble(0x20, 0);           /* now in 4-bit mode */

    lcd_command(LCD_FUNCTION | 0x08);    /* 4-bit, 2 lines, 5x8 */
    lcd_command(LCD_DISPLAY  | 0x00);    /* display off */
    lcd_command(LCD_CLEAR);
    lcd_command(LCD_ENTRY_MODE | 0x02);  /* increment, no shift */
    lcd_command(LCD_DISPLAY  | 0x04);    /* display on, no cursor */
}

static void LCD_set_cursor(uint8_t col, uint8_t row) {
    static const uint8_t row_base[] = { 0x00, 0x40 };
    if (row > 1) row = 1;
    lcd_command((uint8_t)(LCD_SET_DDRAM | (row_base[row] + col)));
}

static void LCD_print(const char *s) {
    while (*s) lcd_data((uint8_t)*s++);
}

/* Load a 5x8 glyph into CGRAM slot 0..7. data[] is 8 rows, low 5 bits used. */
static void LCD_create_char(uint8_t slot, const uint8_t data[8]) {
    slot &= 0x07;
    lcd_command((uint8_t)(LCD_SET_CGRAM | (slot << 3)));
    for (uint8_t i = 0; i < 8; i++) lcd_data(data[i]);
}

/* ============================================================================
 * Demo
 * ========================================================================== */

static const uint8_t heart[8] = {
    0b00000,
    0b01010,
    0b11111,
    0b11111,
    0b11111,
    0b01110,
    0b00100,
    0b00000,
};

int main(void) {
    char line[17];
    uint16_t seconds = 0;

    CCP = CCP_IOREG_gc;
    CLKCTRL.OSCHFCTRLA = CLKCTRL_FRQSEL_24M_gc;

    TWI0_init();
    LCD_init();
    LCD_create_char(0, heart);

    LCD_set_cursor(0, 0);
    LCD_print("AVR128DA28     ");
    lcd_data(0); // heart glyph

    while (1) {
        snprintf(line, sizeof(line), "Uptime:   %5us", seconds);
        LCD_set_cursor(0, 1);
        LCD_print(line);
        seconds++;
        _delay_ms(1000);
    }
}
```

## Compilar y flashear

Para compilarlo deberá ejecutar el siguiente comando:

```bash
avr-gcc -mmcu=avr128da28 \
    -DF_CPU=24000000UL \
    -g -Os -std=gnu99 -Wall -o main.elf *.c
avr-objcopy -O ihex  main.elf main.hex
```

Para subir el programa al microcontrolador, deberá ejecutar el siguiente comando:

```bash
avrdude -c serialupdi -p avr128da28 -P /dev/tty.usbserial-2110 -e -F
```

Si todo fue bien, podrá disfrutar del LCD.
